> [!done]+ 创建新笔记
> 点击 `button-note` 按钮可以创建一个新的笔记。
>
> 2026 年 10 月 21 日更新，版本 1.1.2

```dataviewjs
const { vault, metadataCache } = app;
const current = dv.current().file;
const stripExt = (p) => (p ?? "").replace(/\.md$/i, "");

// 引用信息直接取自 Obsidian 元数据缓存，不解析原文：
// resolvedLinks 已经处理了别名、标题锚点、嵌入 ![[...]] 和 Markdown 链接
const resolved = metadataCache.resolvedLinks[current.path] ?? {};
const unresolved = metadataCache.unresolvedLinks[current.path] ?? {};

const referencedPaths = new Set(Object.keys(resolved).map(stripExt));
const referencedNames = new Set(
    Object.keys(unresolved).map((key) => stripExt(key.split("/").pop()))
);

const isReferenced = (path) =>
    referencedPaths.has(stripExt(path)) ||
    referencedNames.has(stripExt(path.split("/").pop()));

const siblings = vault.getAbstractFileByPath(current.folder || "/")?.children ?? [];
const items = [];

for (const child of siblings) {
    if (child.path === current.path) continue;

    if (child.extension) {
        if (!isReferenced(child.path)) {
            items.push({ name: child.basename, path: child.path });
        }
        continue;
    }

    // 文件夹：优先引用「文件夹/文件夹.md」这样的文件夹笔记
    const folderNote = vault.getAbstractFileByPath(`${child.path}/${child.name}.md`);
    const target = folderNote?.path ?? child.path;
    if (!isReferenced(target)) {
        items.push({ name: child.name, path: target });
    }
}

items.sort((a, b) => a.name.localeCompare(b.name, "zh"));

if (items.length === 0) {
    dv.paragraph("*当前目录下所有文档都已被引用*");
} else {
    dv.list(items.map((item) => dv.fileLink(item.path, false, item.name)));
}
```
